// ==========================================
// CLASS 1: KAMARKOS
// ==========================================
class KamarKos(
    val nomorKamar: String,
    hargaAwal: Int,
    isTersediaAwal: Boolean = true
) {
    // KEAMANAN 1: Custom Setter untuk memvalidasi harga tidak boleh minus
    var harga: Int = hargaAwal
        set(value) {
            if (value < 0) {
                println("⚠️ [Error]: Harga kamar tidak boleh minus! Nilai gagal diubah.")
            } else {
                field = value
            }
        }

    // KEAMANAN 2: Private Set agar status keterisian tidak bisa diubah sembarangan dari luar
    var isTersedia: Boolean = isTersediaAwal
        private set

    fun tampilkanInfo() {
        val status = if (isTersedia) "✅ Kosong" else "❌ Terisi"
        println("==========================================")
        println("           INFORMASI KAMAR: $nomorKamar")
        println("------------------------------------------")
        println(" Harga  : Rp${"%,d".format(harga)}")
        println(" Status : $status")
        println("==========================================\n")
    }

    // Fungsi internal untuk mengubah status secara aman
    fun ubahStatusKamar() {
        isTersedia = !isTersedia
    }
}

// ==========================================
// CLASS 2: TRANSAKSI
// ==========================================
class Transaksi(
    val transaksiId: String,
    val kamar: KamarKos,
    totalBayarAwal: Int
) {
    // KEAMANAN 3: Custom Setter memastikan nominal pembayaran valid
    var totalBayar: Int = totalBayarAwal
        set(value) {
            if (value < 0) {
                println("⚠️ [Error]: Total pembayaran tidak boleh kurang dari 0!")
            } else {
                field = value
            }
        }

    // KEAMANAN 4: Private Set untuk mengamankan status alur transaksi
    var statusPemesanan: String = "PENDING"
        private set 

    fun konfirmasiPembayaran() {
        statusPemesanan = "SUKSES"
        if (kamar.isTersedia) {
            kamar.ubahStatusKamar() // Mengubah status kamar menjadi terisi secara otomatis & aman
        }
    }
    
    fun batalkanTransaksi() {
        statusPemesanan = "BATAL"
        if (!kamar.isTersedia) {
            kamar.ubahStatusKamar() // Mengembalikan status kamar menjadi kosong kembali
        }
    }
}
